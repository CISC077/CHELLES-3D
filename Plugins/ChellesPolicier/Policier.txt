# 1 = agents actifs, 0 = aucun agent
Actif=1

# Personnage utilise : un dossier (le plugin prend le personnage qu'il contient,
# celui qui s'appelle "Agent" en priorite) ou le chemin complet d'un asset.
# Agente de circulation (tenue de police + matraque) :
Modele=/Game/Personnages/POLICES
# Ancien policier (avec gestes des bras) :

# Taille de l'agent en cm (le modele est mis a cette taille)

# Ou poser des agents (1 = oui, 0 = non)
RondsPoints=1
GrandsCarrefours=1

# Un "grand carrefour" = au moins BranchesMinimum rues qui se croisent,
# dont au moins BranchesPrincipales rues a 50 km/h ou plus.
BranchesMinimum=4
BranchesPrincipales=1
VitesseRuePrincipaleKmh=45

# Distance (cm) entre le centre du carrefour et l'agent, qui se tient au coin du trottoir

# Les agents n'apparaissent qu'a moins de cette distance du joueur (cm). 25000 = 250 m
DistanceApparition=25000

# Nombre maximum d'agents en meme temps (les plus proches du joueur)
AgentsMaximum=12

# Animations (facultatif). Laissez vide : le plugin cherche dans DossierAnimations
# des animations du squelette police_officer_Skeleton dont le nom contient
# Repos/Idle (attente), Passez/Wave/Direct/Signal/Point (faire passer), Stop/Halt (arreter).
# Sans animation, l'agent fait les gestes tout seul (bras calcules par le plugin).

AnimRepos=
AnimPassez=
AnimStop=

# Rotation du modele en degres (vide = automatique). A changer seulement si l'agent
# regarde de cote ou de dos : essayez 90, -90 ou 180.
CorrectionRotation=

# Modele fige (Static Mesh) : de quel cote est sa tete (AxeHaut) et vers ou regarde
# son visage (AxeDevant), dans le modele. Valeurs : +X, -X, +Y, -Y, +Z, -Z.
# Agente de Personnages/POLICES : tete en -Y, visage en +Z (mesure sur le modele).
# Si elle est encore couchee ou la tete en bas, essayez AxeHaut=+Y.
AxeHaut=-Y
